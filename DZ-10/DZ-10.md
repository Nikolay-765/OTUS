# ДЗ № 10
## Тема: Управление процессами
Задание: \
Создайте скрипт, который получает информацию о процессах через файловую систему /proc. \
Реализуйте вывод не менее следующих полей: PID, PPID, состояние процесса, имя или команда запуска.

## Выполнение задания
Скрипт №1 (ps-1.sh) \
Простой, в одну строчку. Работает быстро.
```
#!/bin/bash

echo "  PID   PPID  Status   Name"
find /proc -regextype posix-extended -regex "/proc/[0-9]{1,5}/stat" -exec cat {} + | awk '{printf "%5s  %5s     %s     %s\n", $1, $4, $3, $2}'
```

Скрипт №2 (ps-2.sh) \
Посложнее и поитереснее. Работае медленней, но позволяев более гибко получать, обрабатывать и выводить информацию по процессам.
```
#!/bin/bash

# Формируем список дирректорий живых процессов (соотвествует PID) и сортируем их в возрастающем порядке
ProcesList="$(ls '/proc' | grep -wP "[0-9]{1,5}" | tr ' ' '\n' | sort -n | paste -sd ' ' -)"

echo "  PID   PPID  Status       User          TTY           Name"
for ProcessDir in $ProcesList; do
# Проверяем живли ли еще процесс, который с момента формирования списка дирректорий процессов мог умереть
        if [ -d "/proc/$ProcessDir" ]; then
                ProcessInfo="$(awk '{printf "%5s  %5s    %s   ", $1, $4,   $3}' "/proc/$ProcessDir/stat")\t"
                User="$(cat /proc/$ProcessDir/status | grep Uid | awk '{ print $2 }' | xargs id -nu)"
                ProcessInfo="$ProcessInfo$(printf "%-12.12s" "$User")\t"
                if [ -e "/proc/$ProcessDir/fd/0" ] && [ "$(readlink -f "/proc/$ProcessDir/fd/0")" != "/dev/null" ]; then
                        ProcessInfo="$ProcessInfo$(ls -l /proc/$ProcessDir/fd/0 | sed 's|.*/dev/||')\t"
                        Name="$(cat /proc/$ProcessDir/stat | awk '{ print $2 }')"
                        ProcessInfo="$ProcessInfo$(printf "%-30s" "${Name:1:-1}")"
                else
                        ProcessInfo="$ProcessInfo ?\t"
                        Name="$(cat /proc/$ProcessDir/stat | awk '{ print $2 }')"
                        ProcessInfo="$ProcessInfo$(printf "%-30s" "[${Name:1:-1}]")"
                fi
                ProcessInfo="$ProcessInfo\t$(readlink "/proc/$ProcessDir/exe")"
#		Печатаем данные только если процесс жив, за время формирования строки с данными по процессу он также мог отъехать
                if [ -d "/proc/$ProcessDir" ]; then
                        echo -e "$ProcessInfo"
                fi
        fi
done
```
